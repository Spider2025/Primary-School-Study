<script>

// =========👉 修改呢行！填入你嘅Cloudflare Worker網址

const PROXY_URL = "openrouter-proxy.joeng-ncl888.workers.dev";


const modelSel = document.getElementById('modelSel');

const chatArea = document.getElementById('chatArea');

const userInput = document.getElementById('userInput');

const sendBtn = document.getElementById('sendBtn');


function appendMessage(text, role){

  const div = document.createElement('div');

  div.className = 'msg ' + role;

  div.textContent = text;

  chatArea.appendChild(div);

  chatArea.scrollTop = chatArea.scrollHeight;

}


sendBtn.onclick = async ()=>{

  const model = modelSel.value;

  const prompt = userInput.value.trim();

  if(!prompt) return;


  appendMessage(prompt, "user");

  userInput.value = "";

  sendBtn.disabled = true;


  try{

    const resp = await fetch(PROXY_URL,{

      method:"POST",

      headers:{

        "Content-Type":"application/json"

      },

      body: JSON.stringify({

        model: model,

        messages: [{"role":"user","content":prompt}]

      })

    });


    const data = await resp.json();

    if(!resp.ok){

      appendMessage(`錯誤：${JSON.stringify(data)}`,"error");

      return;

    }

    const reply = data.choices?.[0]?.message?.content || "(冇回覆)";

    appendMessage(reply,"assistant");

  }catch(err){

    appendMessage("請求失敗："+err.message,"error");

  }finally{

    sendBtn.disabled = false;

  }

};


userInput.addEventListener('keydown',e=>{

  if(e.key==='Enter' && !e.shiftKey){

    e.preventDefault();

    sendBtn.click();

  }

})

</script>
